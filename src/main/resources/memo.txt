new BukkitRunnable() {
    int timer = 10; // 10回（約5秒間）繰り返す

    @Override
    public void run() {
        if (timer <= 0) {
            this.cancel();
            return;
        }

        // 煙を出す
        loc.getWorld().spawnParticle(Particle.CAMPFIRE_COSY_SMOKE, loc, 30, 1.0, 1.0, 1.0, 0.02);

        // 範囲内のプレイヤーに効果
        loc.getWorld().getNearbyEntities(loc, 3, 3, 3).stream()
           .filter(e -> e instanceof Player)
           .forEach(p -> ((Player)p).addPotionEffect(new PotionEffect(PotionEffectType.BLINDNESS, 40, 1)));

        timer--;
    }
}.runTaskTimer(plugin, 0L, 10L); // 10ティック（0.5秒）おきに実行
